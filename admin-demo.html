async function loadApplications() { if (!currentUser) return; $("applications").textContent = "Applications ਲੋਡ ਹੋ ਰਹੀਆਂ ਹਨ…"; const { data, error } = await db .from("ngsf_member_registrations") .select(` id, user_id, full_name, gender, date_of_birth, father_name, profession, blood_group, state, district, mobile, whatsapp, address, pin_code, email, id_type, status, created_at, updated_at `) .order("created_at", { ascending: false }) .limit(200); if (error) { $("applications").textContent = ""; message("Applications ਲੋਡ ਨਹੀਂ ਹੋਈਆਂ: " + error.message, true); return; } allApplications = data || []; renderApplications(); }
